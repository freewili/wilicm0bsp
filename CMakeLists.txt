cmake_minimum_required(VERSION 3.24)
project(wilicm0bsp VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)

option(WILICM0_BUILD_DRIVER "Build the Linux SPI/UART driver and bridge CLI" OFF)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/libs/onewili/c/CMakeLists.txt")
  message(FATAL_ERROR "Missing OneWili: run git submodule update --init --recursive")
endif()
set(ONEWILI_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(libs/onewili/c)
set(FWCM0_TARGET ${WILICM0_BUILD_DRIVER} CACHE BOOL "" FORCE)
set(FWCM0_BUILD_TESTS ${BUILD_TESTING} CACHE BOOL "" FORCE)
add_subdirectory(drivers/fwcm0)

if(UNIX)
  add_library(wilicm0 STATIC bsp/src/device.cpp)
  target_include_directories(wilicm0 PUBLIC bsp/include)
  target_link_libraries(wilicm0 PUBLIC onewili)
  add_library(wilicm0::wilicm0 ALIAS wilicm0)
  add_subdirectory(apps/device_info)
endif()

if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME app_tools COMMAND ${Python3_EXECUTABLE}
    -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_tools.py)
  add_test(NAME pc_deploy COMMAND ${Python3_EXECUTABLE}
    -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_deploy.py)
  add_test(NAME wifi_analyzer COMMAND ${Python3_EXECUTABLE}
    -m unittest discover -s ${CMAKE_CURRENT_SOURCE_DIR}/tests -p test_wifi_analyzer.py)
  if(UNIX)
    add_executable(socket_probe tests/socket_probe.cpp)
    target_link_libraries(socket_probe PRIVATE wilicm0)
    add_test(NAME bridge_socket COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_socket.py $<TARGET_FILE:socket_probe>)
    set_tests_properties(bridge_socket PROPERTIES TIMEOUT 30)
  endif()
endif()
